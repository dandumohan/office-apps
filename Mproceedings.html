-- Live, always-in-sync aggregate of your existing fund_application table,
-- grouped the same way the Proceedings Builder groups rows. A VIEW needs no
-- trigger and can't fall out of sync — it recomputes on every query.
create or replace view fund_application_summary as
select
  batch_no,
  esti_no,
  token_no,
  sum(amount)            as amount,
  min(application_date)  as application_date,
  count(*)               as task_count
from fund_application
group by batch_no, esti_no, token_no;

-- Your fund_application table already restricts SELECT to the authenticated
-- role via has_perm('Fund Application','V') — this app signs the viewer in
-- with their own Supabase account instead of using the public anon key, so
-- those existing policies apply exactly as they do everywhere else. Just
-- let authenticated users read the view (the policies underneath still gate
-- the real rows per user):
grant select on fund_application_summary to authenticated;

-- Each person who'll use this app needs a Supabase Auth user (Authentication
-- -> Users -> Add user) with 'Fund Application' view permission in your
-- has_perm system, same as for any other tool that reads this table.

-- estimate / advance / withdrawal_accounts are read directly (no view needed
-- -- they're already one row per Esti No / Token No / Account No). FIRST
-- check whether each already has an authenticated-read policy of its own
-- (the same way fund_application does), e.g.:
--   select * from pg_policies where tablename in ('estimate','advance','withdrawal_accounts');
-- Only run the matching create policy below for a table that has NONE yet --
-- adding an unconditional "authenticated can read everything" policy on a
-- table that already has permission-gated policies (like fund_application's
-- has_perm ones) would bypass that gating for anyone merely signed in, not
-- just users with the right view permission.
create policy "authenticated read" on estimate
  for select to authenticated using (true);
create policy "authenticated read" on advance
  for select to authenticated using (true);
create policy "authenticated read" on withdrawal_accounts
  for select to authenticated using (true);

-- Separate table, only for entries typed into "Add missing Fund Application
-- entries" below, for esti/tokens that haven't landed in fund_application
-- yet. Never touched automatically — once the real rows appear in
-- fund_application, fund_application_summary picks them up on its own and
-- this app stops counting the matching manual row to avoid double-counting.
create table if not exists manual_fa_rows (
  id uuid primary key default gen_random_uuid(),
  batch_no text,
  esti_no text not null,
  token_no text,
  amount numeric not null,
  application_date date,
  manual_account_no text,
  manual_vss_name text,
  created_at timestamptz default now()
);

alter table manual_fa_rows enable row level security;

-- Restricted to signed-in users too, so an anon key alone can't write
-- manual entries — consistent with fund_application's own policies.
create policy "authenticated read" on manual_fa_rows
  for select to authenticated using (true);
create policy "authenticated insert" on manual_fa_rows
  for insert to authenticated with check (true);
create policy "authenticated update" on manual_fa_rows
  for update to authenticated using (true) with check (true);
create policy "authenticated delete" on manual_fa_rows
  for delete to authenticated using (true);

-- Shared "Configure the fixed text" settings (letterhead, references,
-- paragraphs, To lists...). One row, id = 'default', saved automatically
-- by the app so every signed-in computer sees the same text.
create table if not exists proceedings_config (
  id text primary key default 'default',
  config jsonb not null,
  updated_at timestamptz default now()
);

alter table proceedings_config enable row level security;

create policy "authenticated read" on proceedings_config
  for select to authenticated using (true);
create policy "authenticated insert" on proceedings_config
  for insert to authenticated with check (true);
create policy "authenticated update" on proceedings_config
  for update to authenticated using (true) with check (true);
