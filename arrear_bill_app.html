Arrear Bill Preparation

Supabase connection (set once)

Run this SQL once in the Supabase SQL editor (safe to re-run):

create table if not exists pay_drawings(
 id uuid primary key default gen_random_uuid(),
 cfms_id text, emp_hrm_id text, pay jsonb, ded jsonb, emp jsonb,
 created_at timestamptz default now());
create table if not exists arrear_bills(
 id uuid primary key default gen_random_uuid(),
 emp_hrm_id text, cfms_id text, payments jsonb, deductions jsonb, net_payable numeric,
 arrear_periods jsonb, total_arrear numeric, created_at timestamptz default now());
alter table arrear_bills add column if not exists pay_id uuid;
alter table arrear_bills add column if not exists details jsonb;
alter table pay_drawings enable row level security;
alter table arrear_bills enable row level security;
drop policy if exists "open" on pay_drawings;
drop policy if exists "open" on arrear_bills;
drop policy if exists "auth_only" on pay_drawings;
drop policy if exists "auth_only" on arrear_bills;
create policy "auth_only" on pay_drawings for all to authenticated using(true) with check(true);
create policy "auth_only" on arrear_bills for all to authenticated using(true) with check(true);
create table if not exists master_data(
 id uuid primary key default gen_random_uuid(),
 sheet text not null, row_no int not null, cells jsonb not null,
 created_at timestamptz default now(), unique(sheet,row_no));
alter table master_data enable row level security;
drop policy if exists "auth_only" on master_data;
create policy "auth_only" on master_data for all to authenticated using(true) with check(true);
notify pgrst, 'reload schema';
